function Run-Job-Visible ([scriptblock]$Code) {
    $Encoded = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($Code.ToString()))
    Start-Process powershell.exe -ArgumentList "-NoProfile", "-ExecutionPolicy", "Bypass", "-EncodedCommand", $Encoded -Wait
}

# --- ACTIONS ---

$BtnTeams.Add_Click({ 
    Run-Job-Visible { 
        Write-Host "--- vi nuker lortet ---" -ForegroundColor Blue
        
        Write-Host "> Lukker alle Office programmer..."
        Stop-Process -Name "winword","excel","powerpnt","outlook","onenote","lync","ms-teams" -Force -ErrorAction SilentlyContinue

        Write-Host "> Fjerner gemte Office Identities i Registry..."
        Remove-Item -Path "HKCU:\Software\Microsoft\Office\16.0\Common\Identity\Identities\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        Write-Host "> Rydder Office Token Cache..."
        $localPath = "$env:LocalAppData\Microsoft\Office\16.0\Licensing"
        Remove-Item -Path "$localPath\*" -Recurse -Force -ErrorAction SilentlyContinue
        
        Write-Host "> Tjekker filer og geninstallerer hvis nødvendigt..."
        $apps = @("WINWORD.EXE", "EXCEL.EXE", "POWERPNT.EXE", "OUTLOOK.EXE")
        $missing = $false
        foreach ($a in $apps) {
            if (!(Test-Path "C:\Program Files\Microsoft Office\root\Office16\$a") -and !(Test-Path "C:\Program Files (x86)\Microsoft Office\root\Office16\$a")) {
                Write-Host "[!] Mangler: $a" -ForegroundColor Red; $missing = $true
            }
        }
        
        if ($missing) {
            Write-Host "> Starter frisk installation via Winget..."
            winget install --id Microsoft.Office -e --silent --accept-package-agreements
        } else {
            Write-Host "> Alle apps fundet. Kontoen er nu nulstillet." -ForegroundColor Green
            Write-Host "> Eleven skal logge ind igen næste gang de åbner Word."
        }

        Read-Host "`nFærdig! Tryk Enter"
    } 
})

# --- ONEDRIVE RESET FRA V64.3 ---
$BtnOneDrive.Add_Click({
    Run-Job-Visible {
        Write-Host "--- ONEDRIVE OPERATION ---" -ForegroundColor Blue
        Stop-Process -Name "OneDrive" -Force -ErrorAction SilentlyContinue
        $paths = @("$env:LocalAppData\Microsoft\OneDrive\OneDrive.exe", "$env:ProgramFiles\Microsoft OneDrive\OneDrive.exe", "$env:ProgramFiles(x86)\Microsoft OneDrive\OneDrive.exe")
        $foundPath = ""; foreach ($p in $paths) { if (Test-Path $p) { $foundPath = $p; break } }
        if ($foundPath -ne "") {
            Write-Host "[OK] Nulstiller OneDrive..." -ForegroundColor Green
            Start-Process $foundPath -ArgumentList "/reset"
        } else {
            Write-Host "[!] Installerer OneDrive..." -ForegroundColor Red
            winget install --id Microsoft.OneDrive -e --silent --accept-package-agreements
        }
        Read-Host "Færdig! Enter"
    }
})

$BtnClean.Add_Click({ Run-Job-Visible { 
    Write-Host "!!! GOD MODE CLEANING !!!" -ForegroundColor Green
    Stop-Service wuauserv, bits, cryptsvc, spooler -Force -ErrorAction SilentlyContinue
    dism /online /cleanup-image /startcomponentcleanup /resetbase
    vssadmin delete shadows /all /quiet
    wevtutil el | Foreach-Object {wevtutil cl "$_"}
    Remove-Item "C:\Windows\SoftwareDistribution\*" -Recurse -Force -ErrorAction SilentlyContinue
    $paths = @("$env:TEMP\*", "C:\Windows\Temp\*", "C:\Windows\Prefetch\*", "C:\Windows\System32\ccmcache\*")
    foreach ($p in $paths) { Remove-Item $p -Recurse -Force -ErrorAction SilentlyContinue }
    ipconfig /flushdns; netsh int ip reset; netsh winsock reset
    Start-Service wuauserv, bits, cryptsvc, spooler -ErrorAction SilentlyContinue
    Read-Host "Færdig! Enter" 
} })

$BtnDebloat.Add_Click({ Run-Job-Visible { 
    Write-Host "--- NUKING BLOAT ---" -ForegroundColor Red

$BtnPerf.Add_Click({ Run-Job-Visible { 
    Write-Host "--- PROCESS OVERBLIK ---" -ForegroundColor Cyan
    Get-Process | Sort-Object CPU -Descending | Select-Object -First 15 @{Name="Program"; Expression={$_.ProcessName}}, @{Name="CPU"; Expression={[math]::Round($_.CPU, 2)}}, @{Name="RAM(MB)"; Expression={[math]::Round($_.WorkingSet / 1MB, 2)}}, @{Name="ID"; Expression={$_.Id}} | Format-Table -AutoSize

} })
